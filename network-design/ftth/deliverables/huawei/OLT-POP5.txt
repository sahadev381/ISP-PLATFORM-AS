enable
config
undo enable alarm output all
sysname OLT-POP5
board confirm 0 frameid 0 slotid 1 boardname GPHF
! --- DBA + profiles (create once, same on every OLT) ---
dba-profile add profile-id 20 profile-name FTTH-RESI type4 max 1000000
dba-profile add profile-id 21 profile-name FTTH-100M type3 assure 50000 max 102400
dba-profile add profile-id 22 profile-name FTTH-50M  type3 assure 25000 max 51200
gpon line-profile add profile-id 20 profile-name FTTH-LINE
  tcont 1 dba-profile-id 20
  gem add 1 eth tcont 1
  gem mapping 1 0 vlan 10
  commit
quit
gpon service-profile add profile-id 20 profile-name FTTH-SVC
  port vlan eth 1 translation 10 user-vlan 10
  commit
quit
! --- Management (VLAN 100) ---
vlan 100 smart
port vlan 100 0/19 *
port vlan 100 0/20 *
interface vlanif 100
  ip address 10.20.5.10 255.255.255.128
quit
ip route-static 0.0.0.0 0.0.0.0 10.20.5.1
! --- Uplink 2x10GE to POP5 L3 switch (LACP trunk) ---
link-aggregation add port 0/19/0 port 0/20/0 mode work lacp-trunk
! --- SNMP/NMS (ISP platform monitoring) ---
snmp-agent sys-info version v2c
snmp-agent community write/privateRO publicRO@FTTH26
snmp-agent target-host trap-hostname NMS address 10.20.0.2 udp-port 162 trap-paramsname NMSNORM
snmp-agent trap enable standard
sntp unicast-server 202.79.35.170
! --- Customer S-VLANs: 1501-1504 ---
vlan 1501 to 1504 smart
port vlan 1501 to 1504 0/1 0-3
port vlan 1501 to 1504 0/19 *
port vlan 1501 to 1504 0/20 *
! --- PON port activation (4 used) ---
interface gpon 0/1
  port 0 ont-auto-find enable
interface gpon 0/1
  port 1 ont-auto-find enable
interface gpon 0/1
  port 2 ont-auto-find enable
interface gpon 0/1
  port 3 ont-auto-find enable
