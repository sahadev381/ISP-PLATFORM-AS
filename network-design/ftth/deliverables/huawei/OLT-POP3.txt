config
undo enable alarm output all

dba-profile add profile-id 21 profile-name FTTH-100M type3 assure 50000 max 102400
dba-profile add profile-id 22 profile-name FTTH-50M  type3 assure 25000 max 51200

  gem add 1 eth tcont 1
  gem mapping 1 0 vlan 10
  commit
quit
gpon service-profile add profile-id 20 profile-name FTTH-SVC
  port vlan eth 1 translation 10 user-vlan 10
  commit
quit
